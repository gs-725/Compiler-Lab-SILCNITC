type
  List
  {
    int data;
    List next;
  }
endtype

decl
    List head;
enddecl

int main()
{
  decl
    int length, x;
    List p[2];
  enddecl

  begin
    initialize();

    head=null;
    read(length);

    p[1]=head;
    while (length!=0)  do
      read(x);
      p[0]=alloc();
      p[0].data=x;
      p[0].next=null;

      if (head==null) then
          head=p[0];
          p[1]=p[0];
      else
          p[1].next=p[0];
          p[1]=p[1].next;
      endif;

      length=length-1;
    endwhile;

    p[0]=head;
    while(p[0]!=null)  do
      write(p[0].data);
      p[0]=p[0].next;
    endwhile;

    return 1;
  end
}
