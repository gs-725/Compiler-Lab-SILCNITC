decl
  int fact(int n);
  int flp;
  int globalcheck();
enddecl
int globalcheck()
{
	begin
		write(flp);
		return 5;
	end
}
int fact(int n)
{
  decl
    int f;
  enddecl
 
  begin
    if(n<=1) then
      f=1;
    else
      f=n*fact(n-1);
    endif;

    return f;
  end
}
int main()
{
  decl
    int a,res;
  enddecl

  begin
  flp=100;
    read(a);
    res = fact(a);
    write(res);
    a=globalcheck();

    return 0;
  end  
}
